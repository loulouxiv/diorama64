cmake_minimum_required(VERSION 3.28)
project(sm64_ar C CXX)
# Objective-C(++) only for the Apple targets (Metal, ARKit, GameController, AppKit); Linux and Android builds are C/C++.
if(APPLE)
  enable_language(OBJC OBJCXX)
endif()
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(cmake/Toolchain.cmake)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_OBJCXX_STANDARD 17)
option(SM64AR_TESTS "Build test-only seams (no tests ship in this tree)" OFF)
add_subdirectory(runtime)
add_subdirectory(games/sm64)
if(SM64AR_VISIONOS)
  # An ImmersiveSpace can only be declared from a SwiftUI App, and CMake builds Swift only with the Xcode generator.
  enable_language(Swift)
  add_subdirectory(apps/visionos)
elseif(APPLE)
  add_subdirectory(apps/macos_harness)
endif()
if(SM64AR_ANDROID)
  add_subdirectory(apps/quest/native)
endif()
if(SM64AR_LINUX)
  add_subdirectory(apps/linux_headless)
  if(SM64AR_HAVE_VULKAN)
    add_subdirectory(apps/linux_window)
  endif()
endif()
